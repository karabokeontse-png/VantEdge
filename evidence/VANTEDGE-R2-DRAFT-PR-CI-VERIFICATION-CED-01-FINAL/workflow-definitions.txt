=== full-ci.yml ===
     1	name: Full CI + W5 Validation
     2	
     3	on:
     4	  push:
     5	    branches: [ main, develop, recovery/reconstructed ]
     6	  pull_request:
     7	    branches: [ main, develop ]
     8	
     9	jobs:
    10	  w5-determinism-check:
    11	    name: W5 100-Run Determinism Validation
    12	    runs-on: ubuntu-latest
    13	    timeout-minutes: 15
    14	
    15	    steps:
    16	      - name: Checkout repository
    17	        uses: actions/checkout@v4
    18	
    19	      - name: Set up JDK 17
    20	        uses: actions/setup-java@v4
    21	        with:
    22	          java-version: '17'
    23	          distribution: 'temurin'
    24	          cache: gradle
    25	
    26	      - name: Fix gradle.properties for CI
    27	        run: |
    28	          sed -i '/org.gradle.java.home/d' gradle.properties
    29	          sed -i '/android.aapt2FromMavenOverride/d' gradle.properties
    30	
    31	      - name: Grant execute permission for gradlew
    32	        run: chmod +x gradlew
    33	
    34	      - name: Compile W5 Sources
    35	        run: ./gradlew compileDebugUnitTestKotlin --no-daemon
    36	
    37	      - name: Run W5 Full Test Suite
    38	        run: ./gradlew testDebugUnitTest --tests "com.vantedge.app.w5.scoring.*" --no-daemon
    39	
    40	      - name: Verify test artifacts exist
    41	        run: |
    42	          echo "=== Test Results ==="
    43	          find app/build/test-results -name "*.xml" | head -5
    44	          echo "=== Test Report ==="
    45	          ls -la app/build/reports/tests/testDebugUnitTest/ 2>/dev/null || echo "Report directory not found"
    46	
    47	      - name: Upload test results
    48	        if: always()
    49	        uses: actions/upload-artifact@v4
    50	        with:
    51	          name: w5-determinism-results
    52	          path: |
    53	            app/build/test-results/
    54	            app/build/reports/
